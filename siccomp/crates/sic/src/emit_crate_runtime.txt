// ── SIC's values, seen from Rust ─────────────────────────────────────────────

use std::marker::PhantomData;
use std::os::raw::{c_char, c_void};

extern "C" {
    fn malloc(size: usize) -> *mut c_void;
    fn free(p: *mut c_void);
}

const FOREIGN: u64 = 1 << 63;

/// A `string` as SIC lays it out: a view with no ownership. A field of a struct is
/// one of these: the struct owns it. Its bytes need not be UTF-8.
#[repr(C)]
#[derive(Clone, Copy)]
pub struct SicStr {
    data: *const c_char,
    size: u64,
    rc: *mut u64,
}

impl SicStr {
    pub fn as_bytes(&self) -> &[u8] {
        if self.size == 0 { return &[]; }
        unsafe { std::slice::from_raw_parts(self.data as *const u8, self.size as usize) }
    }
    pub fn to_str(&self) -> Result<&str, std::str::Utf8Error> { std::str::from_utf8(self.as_bytes()) }
    pub fn to_string_lossy(&self) -> std::borrow::Cow<'_, str> { String::from_utf8_lossy(self.as_bytes()) }
    pub fn len(&self) -> usize { self.size as usize }
    pub fn is_empty(&self) -> bool { self.size == 0 }
}

impl std::fmt::Display for SicStr {
    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result { f.write_str(&self.to_string_lossy()) }
}

/// An owned reference to a SIC `string`: `Clone` and `Drop` count it, as SIC does.
/// Not `Send`: SIC's counts are not atomic.
pub struct SicString { raw: SicStr, _not_send: PhantomData<*const ()> }

impl SicString {
    /// A copy of `bytes` in a block of its own: SIC code may keep it for as long as
    /// it likes, which it could not do with Rust's memory.
    pub fn new(bytes: impl AsRef<[u8]>) -> SicString {
        let b = bytes.as_ref();
        unsafe {
            let block = malloc(8 + b.len() + 1) as *mut u64;
            *block = 1;
            let text = block.add(1) as *mut u8;
            std::ptr::copy_nonoverlapping(b.as_ptr(), text, b.len());
            *text.add(b.len()) = 0;
            SicString { raw: SicStr { data: text as *const c_char, size: b.len() as u64, rc: block }, _not_send: PhantomData }
        }
    }
    unsafe fn from_raw(raw: SicStr) -> SicString { SicString { raw, _not_send: PhantomData } }
    fn raw(&self) -> SicStr { self.raw }
}

impl std::ops::Deref for SicString {
    type Target = SicStr;
    fn deref(&self) -> &SicStr { &self.raw }
}

impl std::fmt::Display for SicString {
    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result { self.raw.fmt(f) }
}

impl Clone for SicString {
    fn clone(&self) -> SicString {
        unsafe { if self.raw.rc as usize > 1 { *self.raw.rc += 1; } }
        SicString { raw: self.raw, _not_send: PhantomData }
    }
}

impl Drop for SicString {
    fn drop(&mut self) {
        let rc = self.raw.rc;
        if rc as usize <= 1 { return; }
        unsafe {
            *rc -= 1;
            let n = *rc;
            if n & !FOREIGN != 0 { return; }
            if n & FOREIGN != 0 {
                let finalize: unsafe extern "C" fn(*mut c_void) = std::mem::transmute(*rc.add(1));
                finalize(rc as *mut c_void);
            } else {
                free(rc as *mut c_void);
            }
        }
    }
}
